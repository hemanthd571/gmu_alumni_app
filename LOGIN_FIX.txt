<?php
header('Content-Type: application/json');
header('Access-Control-Allow-Origin: *');
header('Access-Control-Allow-Methods: POST, OPTIONS');
header('Access-Control-Allow-Headers: Content-Type');

if ($_SERVER['REQUEST_METHOD'] === 'OPTIONS') {
    http_response_code(200);
    exit;
}

// Use the same absolute path that worked for register.php
$configPath = 'C:/xampp_ss/htdocs/alumni/includes/db_config.php';

if (!file_exists($configPath)) {
    http_response_code(500);
    echo json_encode(['success' => false, 'message' => 'Config file not found']);
    exit;
}

require_once $configPath;

if ($_SERVER['REQUEST_METHOD'] !== 'POST') {
    http_response_code(405);
    echo json_encode(['success' => false, 'message' => 'Method not allowed']);
    exit;
}

$input = file_get_contents('php://input');
$data = json_decode($input, true);

if (empty($data['email']) || empty($data['password'])) {
    http_response_code(400);
    echo json_encode(['success' => false, 'message' => 'Email and password are required']);
    exit;
}

try {
    // Fetch user by email including the ROLE and IS_DIRECTOR fields
    // We check for both 'role' column and 'is_director' column to cover all bases
    // We only allow login if the user is ACTIVE (is_active = 1)
    $stmt = $pdo->prepare("SELECT * FROM users WHERE email_id = ? AND is_active = 1");
    $stmt->execute([$data['email']]);
    $user = $stmt->fetch(PDO::FETCH_ASSOC);

    if ($user && password_verify($data['password'], $user['password'])) {
        
        // Generate a simple token (in production use JWT)
        $token = bin2hex(random_bytes(32));
        
        // Update token in DB
        $updateStmt = $pdo->prepare("UPDATE users SET auth_token = ? WHERE id = ?");
        $updateStmt->execute([$token, $user['id']]);

        // Remove sensitive data
        unset($user['password']);
        unset($user['auth_token']);
        
        // Ensure role is set
        if (!isset($user['role'])) {
            $user['role'] = 'alumni'; // default
        }
        
        // Check if director based on role OR is_director column
        $is_director = 0;
        if (
            (isset($user['role']) && strtolower($user['role']) === 'director') ||
            (isset($user['is_director']) && $user['is_director'] == 1)
        ) {
            $is_director = 1;
        }

        // Add calculated field to response
        $user['is_director'] = $is_director;

        echo json_encode([
            'success' => true,
            'message' => 'Login successful',
            'token' => $token,
            'user' => $user
        ]);
    } else {
        http_response_code(401);
        echo json_encode(['success' => false, 'message' => 'Invalid email or password (or account not approved)']);
    }

} catch (PDOException $e) {
    http_response_code(500);
    echo json_encode(['success' => false, 'message' => 'Database error: ' . $e->getMessage()]);
}
?>
